# 07 Doorbell chime: a doorbell button plays a chime, a light by the door blinks, and a second
# button starts and stops rain sounds.
#
# Wiring:
#   doorbell button                -> TRIG1 jack
#   a 12 V light by the door       -> OUT1 + and -
#   a second button                -> TRIG2 jack
#   speaker on SPK; the SD card from the desktop app (chime.mp3 is sound 1, rain.mp3 sound 2),
#   or by hand: /01/001.mp3 and /02/001.mp3
#
#        [doorbell]--TRIG1   OUT1--[door light]   [button]--TRIG2   SPK--[speaker]

effect doorbell:
  wait for trig1
  audio_single = 1     # chime.mp3
  out1 = on
  sleep 0.3
  out1 = off
  sleep 0.3
  out1 = on
  sleep 0.3
  out1 = off

effect rain:
  wait for trig2
  audio_loop = 2       # rain.mp3: loops until the next sound line
  wait for trig2
  audio_stop
