# Doorbell chime for the front porch (firmware/show-script-examples/07).

effect doorbell:
  wait for trig1
  audio_single = 1     # the chime
  out1 = on
  wait 0.3
  out1 = off
  wait 0.3
  out1 = on
  wait 0.3
  out1 = off

effect rain:
  wait for trig2
  audio_loop = 2       # rain: loops until the next sound line
  wait for trig2
  audio_stop
