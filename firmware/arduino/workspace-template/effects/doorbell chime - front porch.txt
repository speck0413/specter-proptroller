# Doorbell chime for the front porch (firmware/show-script-examples/07).

effect doorbell:
  wait for trig1
  audio_single = 1     # chime.mp3
  out1 = on
  sleep 0.3
  out1 = off
  sleep 0.3
  out1 = on
  sleep 0.3
  out1 = off

effect rain:
  wait for trig2
  audio_loop = 2       # rain.mp3: loops until the next sound line
  wait for trig2
  audio_stop
